# Build-time dependencies for creating the standalone binary.
#
# This top-level file is intentionally hash-free so vulnerability scanners
# (for example Snyk) can parse it reliably. Versions remain pinned exactly.
#
# Hash-locked variants are maintained separately in requirements-build.txt.
#
# macOS runtime dependencies (not needed for the Linux binary build):
#   rumps>=0.4.0   — install with: pip install rumps
#
# rumps pulls in pyobjc-core and pyobjc-framework-Cocoa transitively. The tray
# relies on them directly (Foundation.NSObject / NSThread) to marshal AppKit
# work onto the main thread; without them menu rebuilds fall back to running
# in-place, which is what crashed the app and wedged the WindowServer.

# PyInstaller 6.21.0 (source: pyinstaller-6.21.0.tar.gz)
pyinstaller==6.21.0

# altgraph 0.17.5 (source: altgraph-0.17.5-py2.py3-none-any.whl)
# Transitive dependency used by PyInstaller
altgraph==0.17.5

# pyinstaller-hooks-contrib 2026.6 (source: pyinstaller_hooks_contrib-2026.6-py3-none-any.whl)
# Transitive dependency for PyInstaller hooks
pyinstaller-hooks-contrib==2026.6

# setuptools 83.0.0 (source: setuptools-83.0.0-py3-none-any.whl)
# Required by PyInstaller (>=42.0.0)
setuptools==83.0.0

# packaging 26.3 (source: packaging-26.3-py3-none-any.whl)
# Required by PyInstaller (>=22.0)
packaging==26.3

# certifi 2026.7.22 (source: certifi-2026.7.22-py3-none-any.whl)
# Runtime dependency, bundled into the binary. A frozen app ships libssl but
# not the CA store it was built against, so HTTPS (the GitHub update check)
# fails with CERTIFICATE_VERIFY_FAILED on machines that lack the build host's
# certificate path. certifi supplies those certificates explicitly.
certifi==2026.7.22
