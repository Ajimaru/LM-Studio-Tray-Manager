# Build-time dependencies for creating the standalone binary.

# This top-level file is intentionally hash-free so vulnerability scanners
# (for example Snyk) can parse it reliably. Versions remain pinned exactly.
#
# Hash-locked variants are maintained separately in requirements-build.txt.

# macOS runtime dependencies (not needed for the Linux binary build):
#   rumps>=0.4.0   — install with: pip install rumps
#
# rumps pulls in pyobjc-core and pyobjc-framework-Cocoa transitively. The tray
# relies on them directly (Foundation.NSObject / NSThread) to marshal AppKit
# work onto the main thread; without them menu rebuilds fall back to running
# in-place, which is what crashed the app and wedged the WindowServer.

# packaging (source: py3-none-any wheel)
# Required by PyInstaller (>=22.0)
packaging==26.3

# certifi (source: py3-none-any wheel)
